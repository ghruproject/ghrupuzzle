#!/usr/bin/env bash
set -euo pipefail

base_url="https://pub-0095c1ea3d4849fd954f675d08b9dcd6.r2.dev/releases/2026-website-assembly-practice/assembly/practice"

mkdir -p files

for sample in Sample_AP001 Sample_AP002 Sample_AP003 Sample_AP004; do
  curl --fail --location --output "files/${sample}_R1.fastq.gz" \
    "${base_url}/files/${sample}_R1.fastq.gz"
  curl --fail --location --output "files/${sample}_R2.fastq.gz" \
    "${base_url}/files/${sample}_R2.fastq.gz"
done

curl --fail --location --remote-name "${base_url}/sample_sheet.csv"
curl --fail --location --remote-name "${base_url}/checksums.sha256"

echo "Downloaded four read pairs, sample_sheet.csv and checksums.sha256."
echo "Verify the FASTQ files with: sha256sum -c checksums.sha256"
